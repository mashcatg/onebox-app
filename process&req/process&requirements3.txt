    }

6.3 Label Selector (components/LabelSelector.tsx)
typescript'use client'

import { useState, useEffect } from 'react'
import { Check } from 'lucide-react'
import { CustomLabel } from '@/types/email'
import { CustomLabelBadge } from './CustomLabelBadge'

interface Props {
  accountId: string
  messageId: string
  currentLabels: CustomLabel[]
  userId: string
  onLabelsChange: (labels: CustomLabel[]) => void
}

export function LabelSelector({ accountId, messageId, currentLabels, userId, onLabelsChange }: Props) {
  const [allLabels, setAllLabels] = useState<CustomLabel[]>([])
  const [selectedLabelIds, setSelectedLabelIds] = useState<Set<string>>(new Set())
  const [loading, setLoading] = useState(false)

  useEffect(() => {
    fetchAllLabels()
    setSelectedLabelIds(new Set(currentLabels.map(l => l.id)))
  }, [currentLabels])

  const fetchAllLabels = async () => {
    try {
      const response = await fetch('/api/custom-labels')
      const data = await response.json()
      setAllLabels(data.labels || [])
    } catch (error) {
      console.error('Failed to fetch labels:', error)
    }
  }

  const toggleLabel = async (labelId: string) => {
    setLoading(true)
    try {
      const isSelected = selectedLabelIds.has(labelId)
      
      if (isSelected) {
        // Remove label
        await fetch(
          `/api/emails/${accountId}/messages/${messageId}/labels?labelId=${labelId}`,
          { method: 'DELETE' }
        )
        const newIds = new Set(selectedLabelIds)
        newIds.delete(labelId)
        setSelectedLabelIds(newIds)
      } else {
        // Add label
        await fetch(
          `/api/emails/${accountId}/messages/${messageId}/labels`,
          {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ label_ids: [labelId], userId })
          }
        )
        setSelectedLabelIds(new Set([...selectedLabelIds, labelId]))
      }

      // Fetch updated labels
      const response = await fetch(
        `/api/emails/${accountId}/messages/${messageId}/labels`
      )
      const data = await response.json()
      onLabelsChange(data.labels || [])
    } catch (error) {
      console.error('Failed to toggle label:', error)
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className="space-y-2">
      <div className="text-sm font-medium">Apply Labels</div>
      <div className="space-y-1">
        {allLabels.map((label) => (
          <button
            key={label.id}
            onClick={() => toggleLabel(label.id)}
            disabled={loading}
            className="w-full flex items-center justify-between p-2 rounded hover:bg-accent transition-colors"
          >
            <CustomLabelBadge label={label} size="sm" />
            {selectedLabelIds.has(label.id) && (
              <Check className="h-4 w-4 text-primary" />
            )}
          </button>
        ))}
      </div>
    </div>
  )
}
