4.1 Auth Initiation (app/api/auth/[provider]/route.ts)
typescriptimport { NextRequest, NextResponse } from 'next/server'

export async function GET(
  request: NextRequest,
  { params }: { params: { provider: string } }
) {
  const { searchParams } = new URL(request.url)
  const userId = searchParams.get('userId')
  
  if (!userId) {
    return NextResponse.json({ error: 'User ID required' }, { status: 400 })
  }

  const provider = params.provider
  
  // Validate provider
  if (!['google', 'microsoft', 'yahoo', 'imap'].includes(provider)) {
    return NextResponse.json({ error: 'Invalid provider' }, { status: 400 })
  }
  
  // Store userId in state
  const state = Buffer.from(JSON.stringify({ userId, provider })).toString('base64')
  
  const authUrl = `${process.env.NYLAS_API_URI}/v3/connect/auth?` +
    `client_id=${process.env.NEXT_PUBLIC_NYLAS_CLIENT_ID}&` +
    `redirect_uri=${encodeURIComponent(process.env.NEXT_PUBLIC_CALLBACK_URI!)}&` +
    `response_type=code&` +
    `provider=${provider}&` +
    `access_type=online&` +
    `state=${state}`
  
  return NextResponse.redirect(authUrl)
}
4.2 OAuth Callback Page (app/auth/callback/page.tsx)
typescript'use client'

import { useEffect, useState } from 'react'
import { useRouter, useSearchParams } from 'next/navigation'
import { Loader2, CheckCircle, XCircle } from 'lucide-react'

export default function CallbackPage() {
  const router = useRouter()
  const searchParams = useSearchParams()
  const [status, setStatus] = useState<'processing' | 'success' | 'error'>('processing')
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    const handleCallback = async () => {
      const code = searchParams.get('code')
      const state = searchParams.get('state')
      const errorParam = searchParams.get('error')

      if (errorParam) {
        setStatus('error')
        setError(searchParams.get('error_description') || 'Authentication failed')
        return
      }

      if (!code || !state) {
        setStatus('error')
        setError('Missing authentication parameters')
        return
      }

      try {
        const { userId, provider } = JSON.parse(
          Buffer.from(state, 'base64').toString()
        )

        const response = await fetch('/api/auth/exchange', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ code, userId, provider })
        })

        const data = await response.json()

        if (!response.ok) {
          throw new Error(data.error || 'Failed to exchange code')
        }

        setStatus('success')
        setTimeout(() => router.push('/dashboard'), 1500)
      } catch (err) {
        setStatus('error')
        setError(err instanceof Error ? err.message : 'Unknown error')
      }
    }

    handleCallback()
  }, [searchParams, router])

  return (
    <div className="flex items-center justify-center min-h-screen bg-background">
      <div className="text-center p-8 max-w-md">
        {status === 'processing' && (
          <>
            <Loader2 className="h-16 w-16 animate-spin mx-auto mb-4 text-primary" />
            <h2 className="text-xl font-semibold mb-2">Connecting your email account</h2>
            <p className="text-muted-foreground">Please wait while we set up your connection...</p>
          </>
        )}
        
        {status === 'success' && (
          <>
            <CheckCircle className="h-16 w-16 mx-auto mb-4 text-green-500" />
            <h2 className="text-xl font-semibold mb-2">Successfully connected!</h2>
            <p className="text-muted-foreground">Redirecting to your dashboard...</p>
          </>
        )}
        
        {status === 'error' && (
          <>
            <XCircle className="h-16 w-16 mx-auto mb-4 text-destructive" />
            <h2 className="text-xl font-semibold mb-2">Connection failed</h2>
            <p className="text-destructive mb-6">{error}</p>
            <button
              onClick={() => router.push('/dashboard')}
              className="px-6 py-2 bg-primary text-primary-foreground rounded-lg hover:bg-primary/90"
            >
              Back to Dashboard
            </button>
          </>
        )}
      </div>
    </div>
  )
}
4.3 Token Exchange (app/api/auth/exchange/route.ts)
typescriptimport { NextRequest, NextResponse } from 'next/server'
import { supabaseAdmin } from '@/lib/supabase'

export async function POST(request: NextRequest) {
  try {
    const { code, userId, provider } = await request.json()

    if (!code || !userId || !provider) {
      return NextResponse.json(
        { error: 'Missing required parameters' },
        { status: 400 }
      )
    }

    // Exchange code for grant
    const response = await fetch(`${process.env.NYLAS_API_URI}/v3/connect/token`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${process.env.NYLAS_API_KEY}`
      },
      body: JSON.stringify({
        client_id: process.env.NEXT_PUBLIC_NYLAS_CLIENT_ID,
        code: code,
        redirect_uri: process.env.NEXT_PUBLIC_CALLBACK_URI
      })
    })

    if (!response.ok) {
      const errorData = await response.json()
      throw new Error(errorData.error || 'Failed to exchange token')
    }

    const tokenData = await response.json()
    const { grant_id, email } = tokenData

    // Check if this is the first account
    const { data: existingAccounts } = await supabaseAdmin
      .from('email_accounts')
      .select('id')
      .eq('user_id', userId)

    const isPrimary = !existingAccounts || existingAccounts.length === 0

    // Save to Supabase
    const { data, error } = await supabaseAdmin
      .from('email_accounts')
      .upsert({
        user_id: userId,
        grant_id: grant_id,
        email: email,
        provider: provider,
        grant_status: 'valid',
        is_primary: isPrimary,
        last_sync: new Date().toISOString()
      }, {
        onConflict: 'grant_id'
      })
      .select()
      .single()

    if (error) throw error

    return NextResponse.json({ 
      success: true, 
      account: data,
      message: 'Account connected successfully' 
    })
  } catch (error) {
    console.error('Token exchange error:', error)
    return NextResponse.json(
      { error: error instanceof Error ? error.message : 'Failed to connect account' },
      { status: 500 }
    )
  }
}

5.2 Set Primary Account (app/api/accounts/[accountId]/primary/route.ts)
typescriptimport { NextRequest, NextResponse } from 'next/server'
import { supabaseAdmin } from '@/lib/supabase'

export async function POST(
  request: NextRequest,
  { params }: { params: { accountId: string } }
) {
  try {
    const { userId } = await request.json()

    if (!userId) {
      return NextResponse.json({ error: 'User ID required' }, { status: 400 })
    }

    // Unset all primary flags for this user
    await supabaseAdmin
      .from('email_accounts')
      .update({ is_primary: false })
      .eq('user_id', userId)

    // Set new primary
    const { error } = await supabaseAdmin
      .from('email_accounts')
      .update({ is_primary: true })
      .eq('id', params.accountId)
      .eq('user_id', userId)

    if (error) throw error

    return NextResponse.json({ success: true, message: 'Primary account updated' })
  } catch (error) {
    console.error('Error updating primary account:', error)
    return NextResponse.json(
      { error: 'Failed to update primary account' },
      { status: 500 }
    )
  }
}

5.5 Fetch Unified Emails with Custom Labels (app/api/emails/unified/route.ts)
typescriptimport { NextRequest, NextResponse } from 'next/server'
import { supabase } from '@/lib/supabase'
import { nylas } from '@/lib/nylas'

export async function GET(request: NextRequest) {
  const { searchParams } = new URL(request.url)
  const userId = searchParams.get('userId')
  const limit = parseInt(searchParams.get('limit') || '50')
  const unread = searchParams.get('unread')
  const starred = searchParams.get('starred')
  const customLabelId = searchParams.get('customLabelId')

  if (!userId) {
    return NextResponse.json({ error: 'User ID required' }, { status: 400 })
  }

  try {
    // Get all user's accounts
    const { data: accounts, error: accountsError } = await supabase
      .from('email_accounts')
      .select('id, grant_id, email, provider')
      .eq('user_id', userId)
      .eq('grant_status', 'valid')

    if (accountsError) throw accountsError

    if (!accounts || accounts.length === 0) {
      return NextResponse.json({ 
        data: [], 
        total: 0, 
        message: 'No email accounts connected' 
      })
    }

    // If filtering by custom label
    if (customLabelId) {
      const { data: labeledMessages, error: labelError } = await supabase
        .from('message_custom_labels')
        .select('message_id, email_account_id')
        .eq('custom_label_id', customLabelId)
        .in('email_account_id', accounts.map(a => a.id))

      if (labelError) throw labelError

      // Fetch those specific messages
      const messages = await Promise.all(
        labeledMessages.map(async (item) => {
          const account = accounts.find(a => a.id === item.email_account_id)
          if (!account) return null

          try {
            const message = await nylas.messages.find({
              identifier: account.grant_id,
              messageId: item.message_id
            })

            // Get custom labels for this message
            const { data: msgLabels } = await supabase
              .from('message_custom_labels')
              .select(`
                custom_labels (
                  id,
                  name,
                  color
                )
              `)
              .eq('email_account_id', item.email_account_id)
              .eq('message_id', item.message_id)

            return {
              ...message.data,
              account_email: account.email,
              account_provider: account.provider,
              account_id: account.id,
              custom_labels: msgLabels?.map(l => l.custom_labels) || []
            }
          } catch (err) {
            console.error(`Error fetching message ${item.message_id}:`, err)
            return null
          }
        })
      )

      const validMessages = messages.filter(m => m !== null)
      return NextResponse.json({ data: validMessages, total: validMessages.length })
    }

    // Fetch all messages from all accounts
    const allMessages = await Promise.all(
      accounts.map(async (account) => {
        try {
          const queryParams: any = { limit }
          
          if (unread !== null) queryParams.unread = unread === 'true'
          if (starred !== null) queryParams.starred = starred === 'true'
          
          const messages = await nylas.messages.list({
            identifier: account.grant_id,
            queryParams
          })

          // Get message IDs
          const messageIds = messages.data.map((m: any) => m.id)

          if (messageIds.length === 0) return []

          // Get custom labels for these messages
          const { data: customLabels } = await supabase
            .from('message_custom_labels')
            .select(`
              message_id,
              custom_labels (
                id,
                name,
                color
              )
            `)
            .eq('email_account_id', account.id)
            .in('message_id', messageIds)

          // Attach custom labels to messages
          return messages.data.map((msg: any) => ({
            ...msg,
            account_email: account.email,
            account_provider: account.provider,
            account_id: account.id,
            custom_labels: customLabels
              ?.filter(cl => cl.message_id === msg.id)
              .map(cl => cl.custom_labels) || []
          }))
        } catch (err) {
          console.error(`Error fetching from ${account.email}:`, err)
          return []
        }
      })
    )

    // Flatten and sort by date
    const unified = allMessages
      .flat()
      .sort((a: any, b: any) => (b.date || 0) - (a.date || 0))

    return NextResponse.json({ data: unified, total: unified.length })
  } catch (error) {
    console.error('Error fetching unified emails:', error)
    return NextResponse.json(
      { error: 'Failed to fetch emails' },
      { status: 500 }
    )
  }
}
