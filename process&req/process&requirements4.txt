6.5 Email List (components/EmailList.tsx)
typescript'use client'

import { useState, useEffect } from 'react'
import { EmailListItem } from './EmailListItem'
import { Loader2 } from 'lucide-react'

interface Props {
  userId: string
  customLabelId?: string
  onMessageClick: (message: any) => void
}

export function EmailList({ userId, customLabelId, onMessageClick }: Props) {
  const [messages, setMessages] = useState<any[]>([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState<string | null>(null)

  useEffect(() => {
    fetchEmails()
  }, [userId, customLabelId])

  const fetchEmails = async () => {
    setLoading(true)
    setError(null)
    try {
      let url = `/api/emails/unified?userId=${userId}&limit=50`
      if (customLabelId) {
        url += `&customLabelId=${customLabelId}`
      }

      const response = await fetch(url)
      const data = await response.json()
      
      if (!response.ok) {
        throw new Error(data.error || 'Failed to fetch emails')
      }

      setMessages(data.data || [])
    } catch (error) {
      console.error('Failed to fetch emails:', error)
      setError(error instanceof Error ? error.message : 'Failed to fetch emails')
    } finally {
      setLoading(false)
    }
  }

  if (loading) {
    return (
      <div className="flex items-center justify-center p-8">
        <Loader2 className="h-8 w-8 animate-spin text-primary" />
      </div>
    )
  }

  if (error) {
    return (
      <div className="p-8 text-center">
        <p className="text-destructive mb-4">{error}</p>
        <button
          onClick={fetchEmails}
          className="px-4 py-2 bg-primary text-primary-foreground rounded-lg"
        >
          Try Again
        </button>
      </div>
    )
  }

  if (messages.length === 0) {
    return (
      <div className="p-8 text-center text-muted-foreground">
        No emails found
      </div>
    )
  }

  return (
    <div className="divide-y">
      {messages.map((message) => (
        <EmailListItem
          key={message.id}
          message={message}
          onClick={() => onMessageClick(message)}
        />
      ))}
    </div>
  )
}
6.6 Email Viewer (components/EmailViewer.tsx)
typescript'use client'

import { useState, useEffect } from 'react'
import { formatDistanceToNow } from 'date-fns'
import { ArrowLeft, Reply, Forward, Star, Trash2, Tag } from 'lucide-react'
import { CustomLabelBadge } from './CustomLabelBadge'
import { LabelSelector } from './LabelSelector'
import { Button } from '@/components/ui/button'

interface Props {
  accountId: string
  messageId: string
  userId: string
  onBack: () => void
  onReply: () => void
}

export function EmailViewer({ accountId, messageId, userId, onBack, onReply }: Props) {
  const [message, setMessage] = useState<any>(null)
  const [loading, setLoading] = useState(true)
  const [showLabelSelector, setShowLabelSelector] = useState(false)

  useEffect(() => {
    fetchMessage()
  }, [accountId, messageId])

  const fetchMessage = async () => {
    setLoading(true)
    try {
      const response = await fetch(`/api/emails/${accountId}/messages/${messageId}`)
      const data = await response.json()
      setMessage(data.message)
      
      // Mark as read
      if (data.message?.unread) {
        await fetch(`/api/emails/${accountId}/messages/${messageId}`, {
          method: 'PUT',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ unread: false })
        })
      }
    } catch (error) {
      console.error('Failed to fetch message:', error)
    } finally {
      setLoading(false)
    }
  }

  const handleToggleStar = async () => {
    try {
      await fetch(`/api/emails/${accountId}/messages/${messageId}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ starred: !message.starred })
      })
      setMessage({ ...message, starred: !message.starred })
    } catch (error) {
      console.error('Failed to toggle star:', error)
    }
  }

  const handleDelete = async () => {
    if (!confirm('Are you sure you want to delete this email?')) return
    
    try {
      await fetch(`/api/emails/${accountId}/messages/${messageId}`, {
        method: 'DELETE'
      })
      onBack()
    } catch (error) {
      console.error('Failed to delete message:', error)
    }
  }

  if (loading) {
    return <div className="p-8">Loading message...</div>
  }

  if (!message) {
    return <div className="p-8">Message not found</div>
  }

  return (
    <div className="h-full flex flex-col">
      {/* Header */}
      <div className="border-b p-4 flex items-center justify-between">
        <div className="flex items-center gap-2">
          <Button variant="ghost" size="icon" onClick={onBack}>
            <ArrowLeft className="h-5 w-5" />
          </Button>
          <h2 className="text-lg font-semibold truncate">{message.subject || '(No Subject)'}</h2>
        </div>
        
        <div className="flex items-center gap-2">
          <Button variant="ghost" size="icon" onClick={handleToggleStar}>
            <Star
              className={`h-5 w-5 ${
                message.starred ? 'fill-yellow-400 text-yellow-400' : ''
              }`}
            />
          </Button>
          <Button
            variant="ghost"
            size="icon"
            onClick={() => setShowLabelSelector(!showLabelSelector)}
          >
            <Tag className="h-5 w-5" />
          </Button>
          <Button variant="ghost" size="icon" onClick={onReply}>
            <Reply className="h-5 w-5" />
          </Button>
          <Button variant="ghost" size="icon" onClick={handleDelete}>
            <Trash2 className="h-5 w-5" />
          </Button>
        </div>
      </div>

      {/* Label Selector */}
      {showLabelSelector && (
        <div className="border-b p-4 bg-accent/50">
          <LabelSelector
            accountId={accountId}
            messageId={messageId}
            currentLabels={message.custom_labels || []}
            userId={userId}
            onLabelsChange={(labels) => {
              setMessage({ ...message, custom_labels: labels })
            }}
          />
        </div>
      )}

      {/* Message Details */}
      <div className="flex-1 overflow-auto p-6">
        {/* From/To/Date */}
        <div className="mb-6 space-y-2">
          <div className="flex items-center justify-between">
            <div>
              <div className="font-medium">
                {message.from?.[0]?.name || message.from?.[0]?.email}
              </div>
              <div className="text-sm text-muted-foreground">
                {message.from?.[0]?.email}
              </div>
            </div>
            <div className="text-sm text-muted-foreground">
              {formatDistanceToNow(new Date(message.date * 1000), { addSuffix: true })}
            </div>
          </div>
          
          <div className="text-sm text-muted-foreground">
            To: {message.to?.map((t: any) => t.email).join(', ')}
          </div>

          {message.cc && message.cc.length > 0 && (
            <div className="text-sm text-muted-foreground">
              Cc: {message.cc.map((c: any) => c.email).join(', ')}
            </div>
          )}
        </div>

        {/* Custom Labels */}
        {message.custom_labels && message.custom_labels.length > 0 && (
          <div className="mb-4 flex gap-2 flex-wrap">
            {message.custom_labels.map((label: any) => (
              <CustomLabelBadge key={label.id} label={label} />
            ))}
          </div>
        )}

        {/* Email Body */}
        <div
          className="prose max-w-none"
          dangerouslySetInnerHTML={{ __html: message.body }}
        />

        {/* Attachments */}
        {message.attachments && message.attachments.length > 0 && (
          <div className="mt-6 border-t pt-4">
            <h3 className="font-medium mb-2">Attachments</h3>
            <div className="space-y-2">
              {message.attachments.map((att: any) => (
                
                  key={att.id}
                  href={`/api/emails/${accountId}/attachments/${att.id}/download`}
                  className="block p-3 bg-accent rounded-lg hover:bg-accent/80 transition-colors"
                  download
                >
                  <div className="font-medium">{att.filename}</div>
                  <div className="text-sm text-muted-foreground">
                    {(att.size / 1024).toFixed(2)} KB
                  </div>
                </a>
              ))}
            </div>
          </div>
        )}
      </div>
    </div>
  )
}
6.7 Compose Email (components/ComposeEmail.tsx)
typescript'use client'

import { useState } from 'react'
import { X, Send, Sparkles, Loader2 } from 'lucide-react'
import { Button } from '@/components/ui/button'

interface Props {
  accountId: string
  userId: string
  replyTo?: any
  onClose: () => void
  onSent: () => void
}

export function ComposeEmail({ accountId, userId, replyTo, onClose, onSent }: Props) {
  const [to, setTo] = useState(replyTo?.from?.[0]?.email || '')
  const [subject, setSubject] = useState(replyTo ? `Re: ${replyTo.subject}` : '')
  const [body, setBody] = useState('')
  const [cc, setCc] = useState('')
  const [showCc, setShowCc] = useState(false)
  const [sending, setSending] = useState(false)
  const [composing, setComposing] = useState(false)
  const [aiPrompt, setAiPrompt] = useState('')

  const handleSend = async () => {
    if (!to || !subject || !body) {
      alert('Please fill in all required fields')
      return
    }

    setSending(true)
    try {
      const response = await fetch(`/api/emails/${accountId}/send`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          to,
          subject,
          body,
          cc: cc || undefined,
          replyToMessageId: replyTo?.id
        })
      })

      if (!response.ok) {
        throw new Error('Failed to send email')
      }

      onSent()
      onClose()
    } catch (error) {
      console.error('Failed to send email:', error)
      alert('Failed to send email. Please try again.')
    } finally {
      setSending(false)
    }
  }

  const handleAICompose = async () => {
    if (!aiPrompt) return

    setComposing(true)
    try {
      const response = await fetch(`/api/emails/${accountId}/compose`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ prompt: aiPrompt })
      })

      const data = await response.json()
      if (data.message?.body) {
        setBody(data.message.body)
      }
      if (data.message?.subject) {
        setSubject(data.message.subject)
      }
    } catch (error) {
      console.error('Failed to compose with AI:', error)
      alert('Failed to generate email. Please try again.')
    } finally {
      setComposing(false)
      setAiPrompt('')
    }
  }

  return (
    <div className="fixed inset-0 bg-black/50 flex items-center justify-center z-50 p-4">
      <div className="bg-background rounded-lg shadow-xl w-full max-w-3xl max-h-[90vh] flex flex-col">
        {/* Header */}
        <div className="flex items-center justify-between p-4 border-b">
          <h2 className="text-lg font-semibold">
            {replyTo ? 'Reply' : 'New Message'}
          </h2>
          <Button variant="ghost" size="icon" onClick={onClose}>
            <X className="h-5 w-5" />
          </Button>
        </div>

        {/* AI Compose */}
        <div className="p-4 border-b bg-accent/30">
          <div className="flex gap-2">
            <input
              type="text"
              value={aiPrompt}
              onChange={(e) => setAiPrompt(e.target.value)}
              placeholder="Describe the email you want to write..."
              className="flex-1 px-3 py-2 border rounded-lg"
              disabled={composing}
            />
            <Button
              onClick={handleAICompose}
              disabled={composing || !aiPrompt}
            >
              {composing ? (
                <Loader2 className="h-4 w-4 animate-spin" />
              ) : (
                <>
                  <Sparkles className="h-4 w-4 mr-2" />
                  Compose
                </>
              )}
            </Button>
          </div>
        </div>

        {/* Form */}
        <div className="flex-1 overflow-auto p-4 space-y-4">
          <div>
            <label className="block text-sm font-medium mb-1">To</label>
            <input
              type="email"
              value={to}
              onChange={(e) => setTo(e.target.value)}
              className="w-full px-3 py-2 border rounded-lg"
              placeholder="recipient@example.com"
              required
            />
          </div>

          {showCc && (
            <div>
              <label className="block text-sm font-medium mb-1">Cc</label>
              <input
                type="email"
                value={cc}
                onChange={(e) => setCc(e.target.value)}
                className="w-full px-3 py-2 border rounded-lg"
                placeholder="cc@example.com"
              />
            </div>
          )}

          {!showCc && (
            <button
              onClick={() => setShowCc(true)}
              className="text-sm text-primary hover:underline"
            >
              Add Cc
            </button>
          )}

          <div>
            <label className="block text-sm font-medium mb-1">Subject</label>
            <input
              type="text"
              value={subject}
              onChange={(e) => setSubject(e.target.value)}
              className="w-full px-3 py-2 border rounded-lg"
              placeholder="Email subject"
              required
            />
          </div>

          <div>
            <label className="block text-sm font-medium mb-1">Message</label>
            <textarea
              value={body}
              onChange={(e) => setBody(e.target.value)}
              className="w-full px-3 py-2 border rounded-lg min-h-[300px]"
              placeholder="Write your message..."
              required
            />
          </div>
        </div>

        {/* Footer */}
        <div className="p-4 border-t flex justify-end gap-2">
          <Button variant="outline" onClick={onClose}>
            Cancel
          </Button>
          <Button onClick={handleSend} disabled={sending}>
            {sending ? (
              <>
                <Loader2 className="h-4 w-4 mr-2 animate-spin" />
                Sending...
              </>
            ) : (
              <>
                <Send className="h-4 w-4 mr-2" />
                Send
              </>
            )}
          </Button>
        </div>
      </div>
    </div>
  )
}
6.8 Dashboard Layout (app/dashboard/page.tsx)
typescript'use client'

import { useState, useEffect } from 'react'
import { ConnectAccountButton } from '@/components/ConnectAccountButton'
import { EmailList } from '@/components/EmailList'
import { EmailViewer } from '@/components/EmailViewer'
import { ComposeEmail } from '@/components/ComposeEmail'
import { CustomLabelBadge } from '@/components/CustomLabelBadge'
import { Plus, Inbox, Star, Send, Archive, Tag } from 'lucide-react'
import { Button } from '@/components/ui/button'

export default function DashboardPage() {
  const [userId, setUserId] = useState('your-user-id') // Replace with actual user ID from auth
  const [accounts, setAccounts] = useState<any[]>([])
  const [customLabels, setCustomLabels] = useState<any[]>([])
  const [selectedView, setSelectedView] = useState<'inbox' | 'label'>('inbox')
  const [selectedLabelId, setSelectedLabelId] = useState<string | null>(null)
  const [selectedMessage, setSelectedMessage] = useState<any>(null)
  const [showCompose, setShowCompose] = useState(false)
  const [replyToMessage, setReplyToMessage] = useState<any>(null)

  useEffect(() => {
    fetchAccounts()
    fetchCustomLabels()
  }, [userId])

  const fetchAccounts = async () => {
    try {
      const response = await fetch(`/api/accounts?userId=${userId}`)
      const data = await response.json()
      setAccounts(data.accounts || [])
    } catch (error) {
      console.error('Failed to fetch accounts:', error)
    }
  }

  const fetchCustomLabels = async () => {
    try {
      const response = await fetch('/api/custom-labels')
      const data = await response.json()
      setCustomLabels(data.labels || [])
    } catch (error) {
      console.error('Failed to fetch labels:', error)
    }
  }

  const handleMessageClick = (message: any) => {
    setSelectedMessage(message)
  }

  const handleReply = () => {
    setReplyToMessage(selectedMessage)
    setShowCompose(true)
  }

  return (
    <div className="h-screen flex">
      {/* Sidebar */}
      <div className="w-64 border-r bg-sidebar flex flex-col">
        <div className="p-4 border-b">
          <h1 className="text-xl font-bold">OneBox</h1>
        </div>

        {/* Compose Button */}
        <div className="p-4">
          <Button
            onClick={() => {
              setReplyToMessage(null)
              setShowCompose(true)
            }}
            className="w-full"
          >
            <Plus className="h-4 w-4 mr-2" />
            Compose
          </Button>
        </div>

        {/* Navigation */}
        <nav className="flex-1 overflow-auto p-2 space-y-1">
          <button
            onClick={() => {
              setSelectedView('inbox')
              setSelectedLabelId(null)
              setSelectedMessage(null)
            }}
            className={`w-full flex items-center gap-3 px-3 py-2 rounded-lg hover:bg-accent transition-colors ${
              selectedView === 'inbox' && !selectedLabelId ? 'bg-accent' : ''
            }`}
          >
            <Inbox className="h-5 w-5" />
            <span>Inbox</span>
          </button>

          <button className="w-full flex items-center gap-3 px-3 py-2 rounded-lg hover:bg-accent transition-colors">
            <Star className="h-5 w-5" />
            <span>Starred</span>
          </button>

          <button className="w-full flex items-center gap-3 px-3 py-2 rounded-lg hover:bg-accent transition-colors">
            <Send className="h-5 w-5" />
            <span>Sent</span>
          </button>

          {/* Custom Labels */}
          <div className="pt-4">
            <div className="px-3 py-2 text-xs font-semibold text-muted-foreground flex items-center gap-2">
              <Tag className="h-4 w-4" />
              LABELS
            </div>

      <div className="flex-1 flex">

              customLabelId={selectedLabelId || undefined}

          </div>

        {/* Email Viewer */}
        {selectedMessage && accounts.length > 0 && (
          <div className="flex-1">
            <EmailViewer
              accountId={selectedMessage.account_id}
              messageId={selectedMessage.id}
              userId={userId}
              onBack={() => setSelectedMessage(null)}
              onReply={handleReply}
            />
          </div>
        )}
      </div>

      {/* Compose Modal */}
      {showCompose && accounts.length > 0 && (
        <ComposeEmail
          accountId={accounts.find(a => a.is_primary)?.id || accounts[0].id}
          userId={userId}
          replyTo={replyToMessage}
          onClose={() => {
            setShowCompose(false)
            setReplyToMessage(null)
          }}
          onSent={() => {
            setShowCompose(false)
            setReplyToMessage(null)
            // Refresh email list
          }}
        />
      )}
    </div>
  )
}
